// ========================================== // 5. 巴士模組 (KMB、城巴、港鐵巴士) // ========================================== async function updateBuses() { const busDataContainer = document.getElementById('bus-data'); let htmlContent = ''; // 渲染畫面的小工具 const renderRoute = (name, etas) => { let html = `
${name}
`; let count = 0; etas.slice(0, 2).forEach(eta => { if (eta) { count++; // 判斷是否已經是文字 (例如港鐵巴士的 "5分鐘" 或 "即將抵達") const displayStr = (eta.includes('分鐘') || eta.includes('即將') || eta.includes('離開')) ? eta : getMinutesEta(eta); html += `
下班車${displayStr}
`; } }); if (count === 0) { html += `
暫無班次 / 已收車
`; } return html; }; try { // --- 1. 九巴 KMB --- // (seq 是車站序號，目前約略設定為兆康區，若不準確可微調 seq 數字) const kmbRoutes = [ { route: '261', seq: 10, dir: 'O', name: '九巴 261 (往 粉嶺)' }, { route: '62X', seq: 7, dir: 'O', name: '九巴 62X (往 鯉魚門)' }, { route: '258P', seq: 4, dir: 'O', name: '九巴 258P (往 藍田)' } ]; for (const req of kmbRoutes) { try { const res = await fetch(`https://data.etabus.gov.hk/v1/transport/kmb/route-eta/${req.route}/1`); const json = await res.json(); const etas = (json.data || []) .filter(x => x.seq === req.seq && x.dir === req.dir && x.eta) .map(x => x.eta); htmlContent += renderRoute(req.name, etas); } catch(e) { console.error('KMB error', e); } } // --- 2. 城巴 Citybus (鎖定停靠「欣田邨」站, 代碼 003207) --- // 成功加入了城巴 56 號！ const ctbRoutes = [ { route: '50', name: '城巴 50 (往 尖沙咀)' }, { route: '56', name: '城巴 56 (往 上水)' }, { route: 'B3A', name: '城巴 B3A (往 深圳灣)' } ]; for (const req of ctbRoutes) { try { const res = await fetch(`https://rt.data.gov.hk/v1/transport/citybus-nwfb/eta/CTB/003207/${req.route}`); const json = await res.json(); const etas = (json.data || []).filter(x => x.eta).map(x => x.eta); htmlContent += renderRoute(req.name, etas); } catch(e) { console.error('CTB error', e); } } // --- 3. 港鐵巴士 MTR Bus (K54) --- // 已經修復：改用 API 規定的 POST 請求通道 try { const mtrRes = await fetch('https://rt.data.gov.hk/v1/transport/mtr/bus/getSchedule', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ language: 'zh' }) }); const mtrJson = await mtrRes.json(); let k54Etas = []; if (mtrJson.busStopList) { mtrJson.busStopList.forEach(stop => { // 自動搜尋路線為 K54，且站名包含「兆康」的車站 if (stop.routeId === 'K54' && stop.busStopNameCh && stop.busStopNameCh.includes('兆康')) { if (stop.busInfo) { stop.busInfo.forEach(bus => { if (bus.busDepartureTimeDesc) { k54Etas.push(bus.busDepartureTimeDesc); } }); } } }); } htmlContent += renderRoute('港鐵巴士 K54 (兆康站)', k54Etas); } catch(e) { console.error('MTR Bus error', e); htmlContent += `
港鐵巴士 K54 (兆康站)
載入失敗 (港鐵伺服器拒絕)
`; } busDataContainer.innerHTML = htmlContent; } catch (globalError) { busDataContainer.innerHTML = '
巴士模組載入失敗
'; } }
